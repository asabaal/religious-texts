    """Initialize TF-IDF vectors for semantic search"""
    global _initialized, _vectorizer, _tfidf_matrix, _passage_index
    
    # Create TF-IDF vectors
    _vectorizer = TfidfVectorizer(
        max_features=1000,
        stop_words='english',
        ngram_range=(1, 2),
        lowercase=True,
        strip_accents='unicode'
    )
    _tfidf_matrix = _vectorizer.fit_transform(passages)

def search_passages_for_concept(concept: str, top_k: int = 10) -> List[str]:
    """Search for biblical passages using semantic similarity"""
    query_vector = _vectorizer.transform([concept])
    similarities = cosine_similarity(query_vector, _tfidf_matrix).flatten()
    top_indices = np.argsort(similarities)[::-1][:top_k]
    return [ _passage_index[idx] for idx in top_indices if similarities[idx] > 0.1 ]
